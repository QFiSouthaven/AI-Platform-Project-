from typing import Dict, Any, List, Optional
import torch
import logging
from functools import lru_cache

class LLMService:
    def __init__(
        self,
        model_name: str,
        use_gpu: bool = False,
        max_sequence_length: int = 2048,
        temperature: float = 0.7
    ):
        self.model_name = model_name
        self.use_gpu = use_gpu
        self.max_sequence_length = max_sequence_length
        self.temperature = temperature
        self.device = "cuda" if use_gpu and torch.cuda.is_available() else "cpu"
        
        self.tokenizer = self._load_tokenizer()
        self.model = self._load_model()
        self.generation_pipeline = self._create_pipeline()

    def _load_tokenizer(self):
        """Load the tokenizer for the specified model"""
        try:
            return AutoTokenizer.from_pretrained(
                self.model_name,
                padding_side="left"
            )
        except Exception as e:
            logging.error(f"Failed to load tokenizer: {e}")
            raise

    def _load_model(self):
        """Load the LLM model"""
        try:
            model = AutoModelForCausalLM.from_pretrained(
                self.model_name,
                trust_remote_code=True,
                torch_dtype=torch.float16 if self.use_gpu else torch.float32
            )
            return model.to(self.device)
        except Exception as e:
            logging.error(f"Failed to load model: {e}")
            raise

    def _create_pipeline(self):
        """Create a text generation pipeline"""
        return pipeline(
            "text-generation",
            model=self.model,
            tokenizer=self.tokenizer,
            device=0 if self.device == "cuda" else -1
        )

    def generate_code(
        self,
        prompt: str,
        max_length: Optional[int] = None,
        temperature: Optional[float] = None,
        stop_sequences: Optional[List[str]] = None
    ) -> Dict[str, Any]:
        """Generate code from a natural language prompt"""
        try:
            max_length = max_length or self.max_sequence_length
            temperature = temperature or self.temperature
            
            generation_config = {
                "max_length": max_length,
                "temperature": temperature,
                "do_sample": True,
                "top_p": 0.95,
                "top_k": 50,
                "pad_token_id": self.tokenizer.eos_token_id
            }
            
            if stop_sequences:
                generation_config["eos_token_id"] = [
                    self.tokenizer.encode(stop_seq)[0] 
                    for stop_seq in stop_sequences
                ]

            result = self.generation_pipeline(
                prompt,
                **generation_config
            )
            
            return {
                "generated_text": result[0]["generated_text"],
                "prompt": prompt,
                "model": self.model_name,
                "parameters": generation_config
            }
        except Exception as e:
            logging.error(f"Code generation failed: {e}")
            raise

    @lru_cache(maxsize=100)
    def get_model_info(self) -> Dict[str, Any]:
        """Get information about the loaded model"""
        return {
            "model_name": self.model_name,
            "device": self.device,
            "parameters": f"{sum(p.numel() for p in self.model.parameters()):,}",
            "max_sequence_length": self.max_sequence_length,
            "tokenizer_vocab_size": self.tokenizer.vocab_size
        }</pre>

import logging
from app.services.llm_service import LLMService
from app.services.message_bus import MessageBusService

class CodeAgent:
    def __init__(
        self,
        llm_service: LLMService,
        message_bus: MessageBusService
    ):
        self.llm = llm_service
        self.message_bus = message_bus
        self._register_consumers()

    def _register_consumers(self):
        """Register message bus consumers for code generation tasks"""
        self.message_bus.subscribe(
            "code_generation_requests",
            self.handle_generation_request
        )
        self.message_bus.subscribe(
            "code_optimization_requests",
            self.handle_optimization_request
        )

    def handle_generation_request(self, message: Dict[str, Any]):
        """Process incoming code generation requests"""
        try:
            task_id = message.get("task_id")
            prompt = message.get("prompt")
            context = message.get("context", "")
            language = message.get("language", "python")
            
            if not all([task_id, prompt]):
                logging.warning("Invalid generation request format")
                return

            full_prompt = self._build_prompt(prompt, context, language)
            result = self.llm.generate_code(full_prompt)
            
            self.message_bus.publish("code_generation_results", {
                "task_id": task_id,
                "status": "completed",
                "generated_code": result["generated_text"],
                "metadata": {
                    "model": result["model"],
                    "parameters": result["parameters"]
                }
            })
        except Exception as e:
            logging.error(f"Generation request failed: {e}")
            self.message_bus.publish("code_generation_results", {
                "task_id": task_id,
                "status": "failed",
                "error": str(e)
            })

    def handle_optimization_request(self, message: Dict[str, Any]):
        """Process code optimization requests"""
        try:
            task_id = message.get("task_id")
            code = message.get("code")
            optimization_goal = message.get("goal", "performance")
            
            if not all([task_id, code]):
                logging.warning("Invalid optimization request format")
                return

            prompt = self._build_optimization_prompt(code, optimization_goal)
            result = self.llm.generate_code(prompt)
            
            self.message_bus.publish("code_optimization_results", {
                "task_id": task_id,
                "status": "completed",
                "optimized_code": result["generated_text"],
                "original_code": code,
                "optimization_goal": optimization_goal
            })
        except Exception as e:
            logging.error(f"Optimization request failed: {e}")
            self.message_bus.publish("code_optimization_results", {
                "task_id": task_id,
                "status": "failed",
                "error": str(e)
            })

    def _build_prompt(self, prompt: str, context: str, language: str) -> str:
        """Build a structured prompt for code generation"""
        return f"""Below is a request to generate {language} code. 
Context: {context}

Generate complete, functional code that satisfies this request:
{prompt}

The code should be well-structured, documented, and follow best practices for {language}.
Return only the code without additional explanation:"""

    def _build_optimization_prompt(self, code: str, goal: str) -> str:

        }.get(goal, "improvement")
        
        return f"""Optimize the following code with a focus on {optimization_focus}:
{code}

Provide the optimized version of this code with a brief explanation of the changes made.

import logging
import ast
import astor
from sklearn.metrics import accuracy_score, precision_score, recall_score
from app.services.llm_service import LLMService

class EvaluationService:
    def __init__(self, llm_service: LLMService):
        self.llm = llm_service

    def evaluate_code_generation(
        self,
        generated_code: str,
        reference_code: Optional[str] = None,
        test_cases: Optional[List[Dict]] = None,
        language: str = "python"
    ) -> Dict[str, Any]:
        """Evaluate generated code against reference or test cases"""
        try:
            if test_cases:
                return self._evaluate_with_test_cases(
                    generated_code,
                    test_cases,
                    language
                )
            elif reference_code:
                return self._evaluate_against_reference(
                    generated_code,
                    reference_code,
                    language
                )
            else:
                return self._evaluate_with_llm(
                    generated_code,
                    language
                )
        except Exception as e:
            logging.error(f"Evaluation failed: {e}")
            return {
                "status": "failed",
                "error": str(e),
                "generated_code": generated_code
            }

    def _evaluate_with_test_cases(
        self,
        code: str,
        test_cases: List[Dict],
        language: str
    ) -> Dict[str, Any]:
        """Evaluate code by executing test cases"""
        if language != "python":
            return self._evaluate_with_llm(
                code,
                language,
                f"Test cases: {test_cases}"
            )

        try:
            # Dynamically create a test function
            namespace = {}
            exec(code, namespace)
            test_func = namespace.get('solution', None)
            
            if not test_func or not callable(test_func):
                raise ValueError("No callable 'solution' function found in code")

            # Run test cases
            results = []
            for case in test_cases:
                input_data = case["input"]
                expected = case["output"]
                actual = test_func(*input_data)
                results.append(actual == expected)

            accuracy = sum(results) / len(results)
            return {
                "status": "completed",
                "accuracy": accuracy,
                "passed_tests": sum(results),
                "total_tests": len(results),
                "language": language
            }
        except Exception as e:
            logging.error(f"Test case evaluation failed: {e}")
            raise

    def _evaluate_against_reference(
        self,
        generated_code: str,
        reference_code: str,
        language: str
    ) -> Dict[str, Any]:
        """Evaluate by comparing with reference implementation"""
        if language != "python":
            return self._evaluate_with_llm(
                generated_code,
                language,
                f"Reference code: {reference_code}"
            )

        try:
            # Parse ASTs for structural comparison
            gen_ast = ast.parse(generated_code)
            ref_ast = ast.parse(reference_code)
            
            # Compare AST structures
            similarity_score = self._compare_asts(gen_ast, ref_ast)
            
            return {
                "status": "completed",
                "similarity_score": similarity_score,
                "language": language,
                "evaluation_method": "ast_comparison"
            }
        except Exception as e:
            logging.error(f"Reference comparison failed: {e}")
            raise

    def _evaluate_with_llm(
        self,
        code: str,
        language: str,
        context: str = ""
    ) -> Dict[str, Any]:
        """Use LLM to evaluate code quality"""
        prompt = f"""Evaluate the following {language} code:
{code}

{context}

Provide:
1. Functionality score (1-10)
2. Code quality score (1-10)
3. Three specific strengths
4. Three specific weaknesses
5. Improvement suggestions

Evaluation:"""
        
        result = self.llm.generate_code(prompt, temperature=0.3)
        return {
            "status": "completed",
            "evaluation": result["generated_text"],
            "language": language,
            "evaluation_method": "llm_analysis"
        }

    def _compare_asts(self, ast1: ast.AST, ast2: ast.AST) -> float:
        """Compare two ASTs for structural similarity"""
        # Convert ASTs to normalized strings
        str1 = astor.to_source(ast1)
        str2 = astor.to_source(ast2)
        
        # Simple similarity metric (could be enhanced)
        lines1 = str1.splitlines()
        lines2 = str2.splitlines()
        
        common_lines = set(lines1) & set(lines2)
        similarity = len(common_lines) / max(len(lines1), len(lines2))
        
        return similarity</pre>

from typing import Optional, Dict, Any, List
from enum import Enum
from datetime import datetime

class GenerationTaskStatus(str, Enum):
    PENDING = "pending"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"

class CodeGenerationRequest(BaseModel):
    task_id: str
    prompt: str
    context: Optional[str] = None
    language: str = "python"
    parameters: Optional[Dict[str, Any]] = None
    created_at: datetime = datetime.utcnow()

class CodeGenerationResult(BaseModel):
    task_id: str
    status: GenerationTaskStatus
    generated_code: Optional[str] = None
    prompt: Optional[str] = None
    error: Optional[str] = None
    metadata: Optional[Dict[str, Any]] = None
    completed_at: Optional[datetime] = None

class DebuggingRequest(BaseModel):
    task_id: str
    code: str
    error: Optional[str] = None
    language: str = "python"

class DebuggingResult(BaseModel):
    task_id: str
    original_code: str
    fixed_code: str
    explanation: str
    error_context: Optional[str] = None
    language: str
    success: bool
    error: Optional[str] = None

class EvaluationRequest(BaseModel):
    task_id: str
    generated_code: str
    reference_code: Optional[str] = None
    test_cases: Optional[List[Dict[str, Any]]] = None
    language: str = "python"

class EvaluationResult(BaseModel):
    task_id: str
    status: GenerationTaskStatus
    metrics: Optional[Dict[str, Any]] = None
    evaluation: Optional[str] = None
    error: Optional[str] = None</pre>
