    <script>
        // Simple animation on page load
        document.addEventListener('DOMContentLoaded', function() {
            const cards = document.querySelectorAll('.module-card');
            cards.forEach((card, index) => {
                setTimeout(() => {
                    card.style.opacity = '1';
                }, index * 100);
            });
            
            // Animate score bars
            setTimeout(() => {
                document.querySelectorAll('.score-bar').forEach(bar => {
                    bar.style.width = bar.style.width;
                });
            }, 500);
            
            // Add hover effect to diagram elements
            const diagramElements = document.querySelectorAll('.relative > div');
            diagramElements.forEach(el => {
                el.addEventListener('mouseenter', () => {
                    el.style.transform = el.style.transform.replace('scale(1)', 'scale(1.05)');
                });
                el.addEventListener('mouseleave', () => {
                    el.style.transform = el.style.transform.replace('scale(1.05)', 'scale(1)');
                });
            });
        });
    </script>
