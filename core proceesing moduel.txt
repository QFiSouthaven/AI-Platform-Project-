    <script>
        // Animation for processing status
        document.addEventListener('DOMContentLoaded', function() {
            const statusItems = document.querySelectorAll('.tech-stack-item');
            
            statusItems.forEach((item, index) => {
                // Add delay to each item's animation
                item.style.animationDelay = `${index * 0.1}s`;
            });
            
            // Simulate processing status
            const deployButton = document.querySelector('button');
            deployButton.addEventListener('click', function() {
                this.innerHTML = '<i class="fas fa-spinner fa-spin mr-2"></i> Deploying...';
                this.classList.add('animated-pulse');
                
                // Reset after 3 seconds
                setTimeout(() => {
                    this.innerHTML = '<i class="fas fa-check mr-2"></i> Deployed!';
                    this.classList.remove('animated-pulse');
                    this.classList.remove('bg-accent');
                    this.classList.add('bg-green-500');
                    
                    // Reset after another 2 seconds
                    setTimeout(() => {
                        this.innerHTML = '<i class="fas fa-rocket mr-2"></i> Deploy Module';
                        this.classList.remove('bg-green-500');
                        this.classList.add('bg-accent');
                    }, 2000);
                }, 3000);
            });
        });
    </script>
