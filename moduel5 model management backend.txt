                        <pre>from typing import List, Optional
from app.models.ai_model import AIModel, ModelFramework, ModelTaskType
from app.utils.encryption import encrypt_file, decrypt_file
from app.utils.storage import store_model_file
import os
from datetime import datetime

class ModelService:
    def __init__(self, db):
        self.db = db

    async def save_model(
        self,
        name: str,
        framework: ModelFramework,
        task_type: ModelTaskType,
        version: str,
        file_bytes: bytes,
        input_shape: List[int],
        output_shape: List[int],
        description: Optional[str] = None,
        tags: Optional[List[str]] = None,
        encrypt: bool = False
    ) -> AIModel:
        # Store the model file
        file_path = store_model_file(name, version, file_bytes)
        
        # Encrypt if requested
        if encrypt:
            encrypt_file(file_path)
        
        # Create model document
        model = AIModel(
            name=name,
            description=description,
            framework=framework,
            task_type=task_type,
            version=version,
            file_path=file_path,
            input_shape=input_shape,
            output_shape=output_shape,
            tags=tags or [],
            is_encrypted=encrypt,
            metrics={}
        )
        
        # Save to database
        result = await self.db["ai_models"].insert_one(model.dict(by_alias=True))
        model.id = str(result.inserted_id)
        
        return model

    async def load_model(self, model_id: str) -> AIModel:
        model_data = await self.db["ai_models"].find_one({"_id": model_id})
        if not model_data:
            raise ValueError("Model not found")
        
        if model_data["is_encrypted"]:
            decrypt_file(model_data["file_path"])
        
        return AIModel(**model_data)

    async def get_models(
        self,
        framework: Optional[ModelFramework] = None,
        task_type: Optional[ModelTaskType] = None,
        search: Optional[str] = None
    ) -> List[AIModel]:
        query = {}
        if framework:
            query["framework"] = framework
        if task_type:
            query["task_type"] = task_type
        if search:
            query["$or"] = [
                {"name": {"$regex": search, "$options": "i"}},
                {"description": {"$regex": search, "$options": "i"}},
                {"tags": {"$in": [search]}}
            ]
        
        models = []
        async for model in self.db["ai_models"].find(query):
            models.append(AIModel(**model))
        
        return models</pre>

class PluginService:
    def __init__(self, plugins_dir: str):
        self.plugins_dir = plugins_dir
        os.makedirs(plugins_dir, exist_ok=True)
        self.loaded_plugins = {}

    async def install_plugin(self, plugin_file: UploadFile) -> Plugin:
        # Save the plugin file
        file_path = os.path.join(self.plugins_dir, plugin_file.filename)
        with open(file_path, "wb") as f:
            f.write(await plugin_file.read())
        
        # Load the plugin to verify it
        plugin = self._load_plugin(file_path)
        
        # Register the plugin
        self.loaded_plugins[plugin.id] = plugin
        
        return plugin

    def _load_plugin(self, file_path: str) -> Plugin:
        spec = importlib.util.spec_from_file_location(
            f"plugin_{os.path.basename(file_path)}", 
            file_path
        )
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        
        if not hasattr(module, "Plugin"):
            raise ValueError("Plugin file must define a Plugin class")
        
        plugin_class = module.Plugin
        plugin_instance = plugin_class()
        
        return Plugin(
            id=plugin_instance.id,
            name=plugin_instance.name,
            description=plugin_instance.description,
            version=plugin_instance.version,
            file_path=file_path,
            model_types=plugin_instance.model_types,
            created_at=datetime.utcnow()
        )

    def get_plugin(self, plugin_id: str) -> Optional[Plugin]:

    def get_plugins_for_model(self, model_type: str) -> List[Plugin]:

            plugin for plugin in self.loaded_plugins.values()

                    <pre>from cryptography.fernet import Fernet
import os
from typing import Optional

class ModelEncryption:
    def __init__(self, encryption_key: Optional[str] = None):
        if encryption_key:
            if len(encryption_key) != 32:
                raise ValueError("Encryption key must be 32 bytes")
            self.key = encryption_key.encode()
        else:
            self.key = Fernet.generate_key()
        
        self.cipher = Fernet(self.key)

    def encrypt_file(self, file_path: str):
        with open(file_path, 'rb') as f:
            data = f.read()
        
        encrypted_data = self.cipher.encrypt(data)
        
        with open(file_path, 'wb') as f:
            f.write(encrypted_data)

    def decrypt_file(self, file_path: str):
        with open(file_path, 'rb') as f:
            encrypted_data = f.read()
        
        try:
            decrypted_data = self.cipher.decrypt(encrypted_data)
        except Exception as e:
            raise ValueError("Decryption failed - invalid key or corrupted data") from e
        
        with open(file_path, 'wb') as f:
            f.write(decrypted_data)

    def get_key(self) -> str:
        return self.key.decode()</pre>

                <pre>from abc import ABC, abstractmethod
from typing import Any, Dict

class BasePlugin(ABC):
    def __init__(self):
        self.id = "base-plugin"
        self.name = "Base Plugin"
        self.description = "Base plugin class for model extensions"
        self.version = "1.0.0"
        self.model_types = []  # List of model frameworks this plugin supports

    @abstractmethod
    def preprocess(self, input_data: Any, model_metadata: Dict) -> Any:
        """Preprocess input data before model inference"""
        pass

    @abstractmethod
    def postprocess(self, output_data: Any, model_metadata: Dict) -> Any:
        """Postprocess model output before returning to client"""
        pass

    def on_load(self):
        """Called when plugin is loaded"""
        pass

    def on_unload(self):
        """Called when plugin is unloaded"""
        pass</pre>

                <h3 class="text-xl font-semibold mt-6 mb-4">Example Plugin Implementation</h3>
                <pre>from typing import Any, Dict
import numpy as np
from PIL import Image
from io import BytesIO

class ImagePreprocessPlugin:
    def __init__(self):
        self.id = "image-preprocess-v1"
        self.name = "Image Preprocessor"
        self.description = "Preprocesses images for computer vision models"
        self.version = "1.0.0"
        self.model_types = ["tensorflow", "pytorch"]

    def preprocess(self, input_data: bytes, model_metadata: Dict) -> np.ndarray:
        """Convert uploaded image to normalized numpy array"""
        # Load image from bytes
        img = Image.open(BytesIO(input_data))
        
        # Resize to model input shape
        input_shape = model_metadata["input_shape"]
        img = img.resize((input_shape[0], input_shape[1]))
        
        # Convert to numpy array and normalize
        img_array = np.array(img) / 255.0
        
        # Add batch dimension if needed
        if len(img_array.shape) == 3:
            img_array = np.expand_dims(img_array, axis=0)
            
        return img_array

    def postprocess(self, output_data: np.ndarray, model_metadata: Dict) -> Dict:
        """Convert model output to class probabilities"""
        if model_metadata["task_type"] == "classification":
            return {
                "predictions": output_data.tolist(),
                "top_class": int(np.argmax(output_data))
            }
        return {"output": output_data.tolist()}</pre>
