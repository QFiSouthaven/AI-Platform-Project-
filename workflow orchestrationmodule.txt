    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Initialize workflow canvas with sample nodes
            const workflowCanvas = document.getElementById('workflowCanvas');
            
            // Sample task types with different colors
            const taskTypes = [
                { name: 'Input', color: 'bg-blue-500' },
                { name: 'Process', color: 'bg-green-500' },
                { name: 'Validate', color: 'bg-yellow-500' },
                { name: 'Output', color: 'bg-purple-500' }
            ];
            
            // Create sample nodes
            const nodes = [
                { id: 1, type: 0, x: 20, y: 50, name: 'Data Load' },
                { id: 2, type: 1, x: 150, y: 30, name: 'Clean Data' },
                { id: 3, type: 1, x: 150, y: 80, name: 'Validate Schema' },
                { id: 4, type: 2, x: 280, y: 50, name: 'Quality Check' },
                { id: 5, type: 3, x: 410, y: 50, name: 'Export Results' }
            ];
            
            // Draw nodes
            nodes.forEach(node => {
                const nodeElement = document.createElement('div');
                nodeElement.className = `absolute task-node rounded-full flex items-center justify-center text-white cursor-pointer ${taskTypes[node.type].color}`;
                nodeElement.style.left = `${node.x}px`;
                nodeElement.style.top = `${node.y}px`;
                nodeElement.title = node.name;
                nodeElement.dataset.id = node.id;
                
                // Add icon based on type
                const icon = document.createElement('i');
                if (node.type === 0) icon.className = 'fas fa-download text-xs';
                else if (node.type === 1) icon.className = 'fas fa-cogs text-xs';
                else if (node.type === 2) icon.className = 'fas fa-check-circle text-xs';
                else icon.className = 'fas fa-upload text-xs';
                
                nodeElement.appendChild(icon);
                workflowCanvas.appendChild(nodeElement);
                
                // Add node name label
                const label = document.createElement('div');
                label.className = 'absolute text-xs font-medium text-gray-700 whitespace-nowrap';
                label.style.left = `${node.x + 30}px`;
                label.style.top = `${node.y + 6}px`;
                label.textContent = node.name;
                workflowCanvas.appendChild(label);
            });
            
            // Draw connectors (simplified for demo)
            const connectors = [
                { from: 1, to: 2 },
                { from: 1, to: 3 },
                { from: 2, to: 4 },
                { from: 3, to: 4 },
                { from: 4, to: 5 }
            ];
            
            connectors.forEach(conn => {
                const fromNode = nodes.find(n => n.id === conn.from);
                const toNode = nodes.find(n => n.id === conn.to);
                
                if (fromNode && toNode) {
                    const line = document.createElement('div');
                    line.className = 'connector-line';
                    
                    // Calculate position and length (simplified)
                    const startX = fromNode.x + 24;
                    const startY = fromNode.y + 12;
                    const endX = toNode.x;
                    const endY = toNode.y + 12;
                    
                    const length = Math.sqrt(Math.pow(endX - startX, 2) + Math.pow(endY - startY, 2));
                    const angle = Math.atan2(endY - startY, endX - startX) * 180 / Math.PI;
                    
                    line.style.width = `${length}px`;
                    line.style.left = `${startX}px`;
                    line.style.top = `${startY}px`;
                    line.style.transform = `rotate(${angle}deg)`;
                    line.style.transformOrigin = '0 0';
                    
                    workflowCanvas.insertBefore(line, workflowCanvas.firstChild);
                }
            });
            
            // Add hover effect to tech stack icons
            document.querySelectorAll('.tech-icon').forEach(icon => {
                icon.addEventListener('mouseenter', function() {
                    this.querySelector('img').classList.add('animate-pulse');
                });
                
                icon.addEventListener('mouseleave', function() {
                    this.querySelector('img').classList.remove('animate-pulse');
                });
            });
            
            // Simulate progress bar animation
            const progressBar = document.querySelector('.progress-bar');
            let progress = 65;
            setInterval(() => {
                progress = progress + (Math.random() * 2 - 1);
                progress = Math.max(65, Math.min(progress, 85));
                progressBar.style.width = `${progress}%`;
                document.querySelector('.progress-bar + div span:last-child').textContent = `${Math.round(progress)}%`;
            }, 2000);
        });
    </script>
