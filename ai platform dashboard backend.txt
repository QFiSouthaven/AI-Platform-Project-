const { validateModel } = require('../utils/validation');

exports.uploadModel = async (req, res) => {
    try {
        await uploadModel(req, res);
        
        if (!req.file) {
            return res.status(400).json({ message: 'Please upload a model file' });
        }

        const { error } = validateModel(req.body);
        if (error) {
            fs.unlinkSync(req.file.path);
            return res.status(400).json({ message: error.details[0].message });
        }

        const model = new Model({
            ...req.body,
            filePath: req.file.path,
            createdBy: req.user.id
        });

        await model.save();
        res.status(201).json(model);
    } catch (err) {
        if (req.file) fs.unlinkSync(req.file.path);
        res.status(500).json({ message: err.message });
    }
};

exports.getModels = async (req, res) => {
    try {
        const { framework, taskType, search } = req.query;
        const query = {};
        
        if (framework) query.framework = framework;
        if (taskType) query.taskType = taskType;
        if (search) query.name = { $regex: search, $options: 'i' };
        
        const models = await Model.find(query)
            .populate('createdBy', 'name email')
            .sort({ createdAt: -1 });
            
        res.json(models);
    } catch (err) {
        res.status(500).json({ message: err.message });
    }
};

exports.getModelMetrics = async (req, res) => {

                        <pre>const Docker = require('dockerode');
const fs = require('fs');
const path = require('path');
const logger = require('../utils/logger');
const Deployment = require('../models/Deployment');

const docker = new Docker();

exports.deployModel = async (deploymentId, modelPath) => {
    try {
        const deployment = await Deployment.findById(deploymentId);
        if (!deployment) {
            throw new Error('Deployment not found');
        }

        // Create Docker container with the model
        const container = await docker.createContainer({
            Image: 'tensorflow/serving',
            Cmd: [`--model_name=${deployment.name}`, `--model_base_path=/models`],
            HostConfig: {
                Binds: [`${path.resolve(modelPath)}:/models/${deployment.name}`]
            },
            ExposedPorts: { '8501/tcp': {} },
            PortBindings: { '8501/tcp': [{ HostPort: '0' }] }
        });

        await container.start();
        
        // Get assigned port
        const containerInfo = await container.inspect();
        const port = containerInfo.NetworkSettings.Ports['8501/tcp'][0].HostPort;
        
        // Update deployment status
        deployment.status = 'Active';
        deployment.endpoint = `http://localhost:${port}/v1/models/${deployment.name}:predict`;
        await deployment.save();
        
        logger.info(`Deployment ${deploymentId} successful. Endpoint: ${deployment.endpoint}`);
    } catch (err) {
        logger.error(`Deployment failed: ${err.message}`);
        
        // Update deployment status if it exists

const User = require('../models/User');
const logger = require('../utils/logger');

exports.authenticate = async (req, res, next) => {
    try {
        const token = req.header('Authorization')?.replace('Bearer ', '');
        if (!token) {
            return res.status(401).json({ message: 'No token provided' });
        }
        
        const decoded = jwt.verify(token, process.env.JWT_SECRET);
        const user = await User.findById(decoded.id);
        
        if (!user) {
            return res.status(401).json({ message: 'User not found' });
        }
        
        req.user = user;
        next();
    } catch (err) {
        logger.error(`Authentication error: ${err.message}`);
        res.status(401).json({ message: 'Invalid token' });
    }
};

exports.authorize = (roles) => {
    return (req, res, next) => {
        if (!roles.includes(req.user.role)) {
            return res.status(403).json({ message: 'Unauthorized' });
        }
        next();
    };
